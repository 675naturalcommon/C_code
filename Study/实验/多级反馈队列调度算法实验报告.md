## （3）实验内容

### 内容描述

**解题思路**:
多级反馈队列调度算法是一种结合了多种调度策略的算法，主要特点包括：
1. 系统维护多个优先级不同的队列，优先级从高到低排列
2. 新进程进入最高优先级队列
3. 每个队列分配不同的时间片大小，优先级越高的队列时间片越小
4. 进程在当前队列的时间片用完后，降级到下一级队列
5. 最后一级队列中的进程时间片耗尽时，返回本级队列继续排队
6. 最后一级队列中的进程如果连续被调度两次仍未完成，则返回第一级队列

**所使用的数据结构原理**:
1. **自定义链表节点（ListNode）**: 用于实现队列的基本存储单元
2. **自定义队列（MyQueue）**: 基于链表实现的先进先出（FIFO）数据结构
3. **自定义动态数组（MyVector）**: 实现动态扩容的数组，用于存储进程和队列
4. **进程结构体（Process）**: 封装进程的所有属性和方法
5. **调度器类（MLFQScheduler）**: 封装整个调度算法的核心逻辑

#### 2. 自定义队列实现
```cpp
template<typename T>
class MyQueue {
private:
    ListNode<T>* front;    // 队首指针
    ListNode<T>* rear;     // 队尾指针
    int count;             // 队列元素计数

public:
    MyQueue() : front(nullptr), rear(nullptr), count(0) {}  // 构造函数

    void push(T val) {
        ListNode<T>* newNode = new ListNode<T>(val);  // 创建新节点
        if (rear == nullptr) {  // 队列为空
            front = rear = newNode;
        } else {  // 队列非空，添加到队尾
            rear->next = newNode;
            rear = newNode;
        }
        count++;
    }

    void pop() {
        if (empty()) return;
        ListNode<T>* temp = front;
        front = front->next;  // 移动队首指针
        if (front == nullptr) {
            rear = nullptr;   // 队列变为空
        }
        delete temp;          // 释放内存
        count--;
    }

    T front_val() {
        if (empty()) {
            static T defaultVal;
            return defaultVal;  // 返回默认值（空队列情况）
        }
        return front->data;     // 返回队首元素
    }

    bool empty() const { return front == nullptr; }
    int size() const { return count; }
};
```

#### 4. 核心调度逻辑
```cpp
void schedule() {
    while (completedProcesses < totalProcesses && currentTime < maxTime) {
        // 添加当前时间到达的进程
        addArrivingProcesses();

        // 查找最高优先级非空队列
        int queueIndex = findHighestPriorityNonEmptyQueue();

        if (queueIndex == -1) {
            // 所有队列为空，时间推进
            currentTime++;
            continue;
        }

        // 从队列中取出进程
        Process* currentProcess = queues[queueIndex].front_val();
        queues[queueIndex].pop();

        // 获取当前队列的时间片
        int timeSlice = timeSlices[queueIndex];

        // 确定实际执行时间
        int executionTime = (timeSlice < currentProcess->remainingTime) ?
                           timeSlice : currentProcess->remainingTime;

        // 更新连续调度次数（如果是最后一级队列）
        if (queueIndex == 4) {
            currentProcess->consecutiveSchedules++;
        } else {
            currentProcess->consecutiveSchedules = 0;
        }

        // 执行进程
        currentTime += executionTime;
        currentProcess->remainingTime -= executionTime;
        currentProcess->timeSpentInCurrentQueue += executionTime;

        // 检查进程是否完成
        if (currentProcess->remainingTime == 0) {
            // 进程完成
            currentProcess->finishTime = currentTime;
            currentProcess->completed = true;
            completedProcesses++;

            displayProcessDetails(currentProcess, "进程离开系统");
            displayQueueStatus();
        } else {
            // 进程未完成，需要重新加入队列
            if (currentProcess->timeSpentInCurrentQueue >= timeSlice) {
                // 当前队列时间片已用完
                if (queueIndex == 4) {  // 最后一级队列
                    if (currentProcess->consecutiveSchedules >= 2) {
                        // 特殊规则：连续调度两次仍未完成，返回第一级队列
                        currentProcess->currentQueueLevel = 0;
                        currentProcess->timeSpentInCurrentQueue = 0;
                        currentProcess->consecutiveSchedules = 0;
                        queues[0].push(currentProcess);
                    } else {
                        // 留在最后一级队列
                        currentProcess->timeSpentInCurrentQueue = 0;
                        queues[4].push(currentProcess);
                    }
                } else {
                    // 降级到下一级队列
                    int nextLevel = (queueIndex + 1 < 4) ? queueIndex + 1 : 4;
                    currentProcess->currentQueueLevel = nextLevel;
                    currentProcess->timeSpentInCurrentQueue = 0;
                    currentProcess->consecutiveSchedules = 0;
                    queues[nextLevel].push(currentProcess);
                }
            } else {
                // 时间片未用完，重新加入当前队列
                queues[queueIndex].push(currentProcess);
            }
            displayQueueStatus();
        }
    }
}
```

// 自己实现的队列（使用链表）
template<typename T>
class MyQueue {
private:
    ListNode<T>* front;    // 队首指针
    ListNode<T>* rear;     // 队尾指针
    int count;             // 队列元素计数

public:
    MyQueue() : front(nullptr), rear(nullptr), count(0) {}  // 构造函数

    ~MyQueue() {
        while (!empty()) {
            pop();  // 析构时释放所有节点内存
        }
    }

    void push(T val) {
        ListNode<T>* newNode = new ListNode<T>(val);  // 创建新节点
        if (rear == nullptr) {  // 队列为空
            front = rear = newNode;
        } else {  // 队列非空，添加到队尾
            rear->next = newNode;
            rear = newNode;
        }
        count++;
    }

    void pop() {
        if (empty()) return;
        ListNode<T>* temp = front;
        front = front->next;  // 移动队首指针
        if (front == nullptr) {
            rear = nullptr;   // 队列变为空
        }
        delete temp;          // 释放内存
        count--;
    }

    T front_val() {
        if (empty()) {
            static T defaultVal;
            return defaultVal;  // 返回默认值（空队列情况）
        }
        return front->data;     // 返回队首元素
    }

    bool empty() const {
        return front == nullptr;
    }

    int size() const {
        return count;
    }

    // 复制队列内容（用于遍历）
    MyQueue<T> copy() const {
        MyQueue<T> result;
        ListNode<T>* current = front;
        while (current != nullptr) {
            result.push(current->data);
            current = current->next;
        }
        return result;
    }
};

// 自己实现的简单动态数组
template<typename T>
class MyVector {
private:
    T* data;        // 数据指针
    int capacity;   // 数组容量
    int length;     // 当前长度

    void resize(int newCapacity) {
        T* newData = new T[newCapacity];  // 分配新内存
        for (int i = 0; i < length; i++) {
            newData[i] = data[i];  // 复制数据
        }
        delete[] data;     // 释放旧内存
        data = newData;    // 更新指针
        capacity = newCapacity;  // 更新容量
    }

public:
    MyVector() : data(new T[10]), capacity(10), length(0) {}  // 默认构造函数

    ~MyVector() {
        delete[] data;  // 析构时释放内存
    }

    void push_back(const T& val) {
        if (length >= capacity) {
            resize(capacity * 2);  // 容量不足时扩容
        }
        data[length++] = val;  // 添加元素并更新长度
    }

    T& operator[](int index) {
        return data[index];  // 返回引用，支持修改
    }

    const T& operator[](int index) const {
        return data[index];  // 返回常量引用
    }

    int size() const {
        return length;  // 返回当前长度
    }

    bool empty() const {
        return length == 0;  // 判断是否为空
    }

    void clear() {
        length = 0;  // 清空数组（不释放内存）
    }

    // 简单的查找函数
    bool contains(const T& val) const {
        for (int i = 0; i < length; i++) {
            if (data[i] == val) {
                return true;  // 找到元素
            }
        }
        return false;  // 未找到元素
    }

    // 简单的排序（冒泡排序）
    void sort(bool (*compare)(const T&, const T&)) {
        for (int i = 0; i < length - 1; i++) {
            for (int j = 0; j < length - i - 1; j++) {
                if (!compare(data[j], data[j + 1])) {
                    T temp = data[j];
                    data[j] = data[j + 1];
                    data[j + 1] = temp;  // 交换元素
                }
            }
        }
    }
};

// 多级反馈队列调度器类
class MLFQScheduler {
private:
    MyVector<MyQueue<Process*>> queues; // 5级队列
    MyVector<int> timeSlices;         // 每级队列的时间片大小
    MyVector<Process> processes;      // 所有进程
    int currentTime;                // 当前系统时间
    int totalProcesses;             // 总进程数
    int completedProcesses;         // 已完成进程数

    // 简单随机数生成器
    unsigned int seed;

public:
    MLFQScheduler() : currentTime(0), totalProcesses(0), completedProcesses(0) {
        // 初始化随机数种子
        seed = time(nullptr);

        // 初始化5级队列
        for (int i = 0; i < 5; i++) {
            queues.push_back(MyQueue<Process*>());
        }

        // 生成5个不同的时间片大小 (1~16之间)
        generateTimeSlices();

        // 生成至少5个进程
        generateProcesses(5 + (rand() % 3)); // 5-7个进程
    }

    // 简单随机数函数
    int my_rand(int min, int max) {
        seed = seed * 1103515245 + 12345;
        return min + (seed % (max - min + 1));
    }

    // 检查数组中是否包含某个值
    bool contains(const MyVector<int>& vec, int val) {
        for (int i = 0; i < vec.size(); i++) {
            if (vec[i] == val) {
                return true;
            }
        }
        return false;
    }

    // 生成5个不同的时间片大小
    void generateTimeSlices() {
        MyVector<int> used;

        for (int i = 0; i < 5; i++) {
            int ts;
            do {
                ts = my_rand(1, 16);
            } while (contains(used, ts));

            used.push_back(ts);
            timeSlices.push_back(ts);
        }

        // 确保第一级队列优先级最高（时间片最小）- 使用冒泡排序
        for (int i = 0; i < timeSlices.size() - 1; i++) {
            for (int j = 0; j < timeSlices.size() - i - 1; j++) {
                if (timeSlices[j] > timeSlices[j + 1]) {
                    int temp = timeSlices[j];
                    timeSlices[j] = timeSlices[j + 1];
                    timeSlices[j + 1] = temp;
                }
            }
        }
    }

    // 生成随机进程
    void generateProcesses(int count) {
        for (int i = 0; i < count; i++) {
            // 转换为字符串
            char nameBuf[10];
            sprintf(nameBuf, "P%d", i + 1);
            string name = nameBuf;

            int pid = my_rand(0, 3500);
            int arrive = my_rand(0, 10); // 改为0~10，便于测试
            int burst = my_rand(5, 30);  // 改为5~30，便于测试

            processes.push_back(Process(name, pid, arrive, burst));
            totalProcesses++;
        }

        // 按到达时间排序 - 使用冒泡排序
        for (int i = 0; i < processes.size() - 1; i++) {
            for (int j = 0; j < processes.size() - i - 1; j++) {
                if (processes[j].arriveTime > processes[j + 1].arriveTime) {
                    Process temp = processes[j];
                    processes[j] = processes[j + 1];
                    processes[j + 1] = temp;
                }
            }
        }

        // 打印生成的进程信息用于调试
        cout << "\n生成的进程信息:" << endl;
        for (int i = 0; i < processes.size(); i++) {
            const Process& p = processes[i];
            cout << p.name << ": PID=" << p.pid << ", 到达时间=" << p.arriveTime
                 << ", 服务时间=" << p.burstTime << endl;
        }
    }

    // 显示系统状态
    void displaySystemStatus() {
        cout << "\n" << string(100, '=') << endl;
        cout << "系统初始化完成 - 当前时间: " << currentTime << endl;
        cout << string(100, '=') << endl;

        cout << "\n队列配置:" << endl;
        cout << "级别\t优先级\t时间片大小" << endl;
        cout << "----\t------\t----------" << endl;
        for (int i = 0; i < 5; i++) {
            cout << i + 1 << "\t" << (i + 1) << "\t" << timeSlices[i] << endl;
        }

        cout << "\n进程基本信息:" << endl;
        cout << "名称\tPID\t到达时间\t服务时间\t剩余时间\t当前队列" << endl;
        cout << "----\t---\t--------\t--------\t--------\t--------" << endl;
        for (int i = 0; i < processes.size(); i++) {
            const Process& p = processes[i];
            cout << p.name << "\t" << p.pid << "\t" << p.arriveTime

        }

        displayQueueStatus();
    }

    void displayQueueStatus() {
        cout << "\n各队列状态:" << endl;
        for (int i = 0; i < 5; i++) {
            cout << "队列" << (i + 1) << " (优先级:" << (i + 1)
                 << ", 时间片:" << timeSlices[i] << "): ";

            if (queues[i].empty()) {
                cout << "空" << endl;
            } else {
                cout << queues[i].size() << "个进程 - ";

                // 复制队列以遍历
                MyQueue<Process*> temp = queues[i].copy();
                int pos = 1;
                while (!temp.empty()) {
                    Process* p = temp.front_val();
                    cout << p->name << "(位置:" << pos++ << ",剩余:" << p->remainingTime << ") ";
                    temp.pop();
                }
                cout << endl;
            }
        }
    }

    // 显示进程详细信息
    void displayProcessDetails(Process* p, const string& event) {
        cout << "\n" << string(80, '-') << endl;
        cout << "事件: " << event << " - " << p->getInfo() << endl;
        cout << "当前队列: " << (p->currentQueueLevel + 1)
             << ", 连续调度次数: " << p->consecutiveSchedules << endl;
        cout << string(80, '-') << endl;
    }

    // 将到达的进程加入相应队列
    void addArrivingProcesses() {
        for (int idx = 0; idx < processes.size(); idx++) {
            Process& p = processes[idx];
            if (p.arriveTime == currentTime && !p.completed) {
                // 检查进程是否已经在队列中
                bool alreadyInQueue = false;
                for (int i = 0; i < 5; i++) {
                    MyQueue<Process*> temp = queues[i].copy();
                    while (!temp.empty()) {
                        if (temp.front_val() == &p) {
                            alreadyInQueue = true;
                            break;
                        }
                        temp.pop();
                    }
                    if (alreadyInQueue) break;
                }

                if (!alreadyInQueue) {
                    // 新进程到达，加入最高优先级队列
                    p.currentQueueLevel = 0;
                    p.timeSpentInCurrentQueue = 0;
                    p.consecutiveSchedules = 0;
                    queues[0].push(&p);

                    displayProcessDetails(&p, "新进程进入系统");
                    displayQueueStatus();
                }
            }
        }
    }

    // 查找最高优先级非空队列
    int findHighestPriorityNonEmptyQueue() {
        for (int i = 0; i < 5; i++) {
            if (!queues[i].empty()) {
                return i;
            }
        }
        return -1; // 所有队列都为空
    }

    // 执行调度
    void schedule() {
        cout << "\n" << string(100, '=') << endl;
        cout << "开始多级反馈队列调度模拟" << endl;
        cout << string(100, '=') << endl;

        displaySystemStatus();

        int maxTime = 1000; // 最大时间限制，防止无限循环

        while (completedProcesses < totalProcesses && currentTime < maxTime) {
            // 添加当前时间到达的进程
            addArrivingProcesses();

            // 查找最高优先级非空队列
            int queueIndex = findHighestPriorityNonEmptyQueue();

            if (queueIndex == -1) {
                // 所有队列为空，时间推进
                cout << "\n时间 " << currentTime << ": 所有队列为空，时间推进..." << endl;
                currentTime++;
                continue;
            }

            // 从队列中取出进程
            Process* currentProcess = queues[queueIndex].front_val();
            queues[queueIndex].pop();

            // 记录开始时间
            if (currentProcess->startTime == -1) {
                currentProcess->startTime = currentTime;
            }

            // 获取当前队列的时间片
            int timeSlice = timeSlices[queueIndex];

            // 确定实际执行时间
            int executionTime = (timeSlice < currentProcess->remainingTime) ? timeSlice : currentProcess->remainingTime;

            // 更新连续调度次数（如果是最后一级队列）
            if (queueIndex == 4) {
                currentProcess->consecutiveSchedules++;
            } else {
                // 如果不是最后一级，重置连续调度次数
                currentProcess->consecutiveSchedules = 0;
            }

            // 显示调度信息
            cout << "\n时间 " << currentTime << ": 调度 " << currentProcess->name
                 << " 从队列" << (queueIndex + 1)
                 << " 执行 " << executionTime << " 单位时间" << endl;

            // 执行进程
            currentTime += executionTime;
            currentProcess->remainingTime -= executionTime;
            currentProcess->timeSpentInCurrentQueue += executionTime;

            // 添加在此期间到达的新进程（逐个时间单位添加）
            for (int t = currentTime - executionTime + 1; t <= currentTime; t++) {
                int savedTime = currentTime;
                currentTime = t;
                addArrivingProcesses();
                currentTime = savedTime;
            }

            // 检查进程是否完成
            if (currentProcess->remainingTime == 0) {
                // 进程完成
                currentProcess->finishTime = currentTime;
                currentProcess->completed = true;
                completedProcesses++;

                displayProcessDetails(currentProcess, "进程离开系统");
                displayQueueStatus();

                // 显示完成信息
                cout << "\n✓ 进程 " << currentProcess->name << " 已完成!" << endl;
                cout << "  开始时间: " << currentProcess->startTime
                     << ", 完成时间: " << currentProcess->finishTime
                     << ", 周转时间: " << (currentProcess->finishTime - currentProcess->arriveTime) << endl;
            } else {
                // 进程未完成，需要重新加入队列
                if (currentProcess->timeSpentInCurrentQueue >= timeSlice) {
                    // 当前队列时间片已用完

                    // 检查最后一级队列的特殊规则
                    if (queueIndex == 4) {
                        if (currentProcess->consecutiveSchedules >= 2) {
                            // 最后一级队列中连续调度两次仍未完成，返回第一级队列
                            cout << "⚠️  特殊规则触发: " << currentProcess->name
                                 << " 在最后一级队列中连续调度" << currentProcess->consecutiveSchedules
                                 << "次仍未完成，返回第一级队列!" << endl;

                            currentProcess->currentQueueLevel = 0;
                            currentProcess->timeSpentInCurrentQueue = 0;
                            currentProcess->consecutiveSchedules = 0;
                            queues[0].push(currentProcess);
                        } else {
                            // 留在最后一级队列
                            currentProcess->timeSpentInCurrentQueue = 0;
                            queues[4].push(currentProcess);
                        }
                    } else {
                        // 降级到下一级队列
                        int nextLevel = (queueIndex + 1 < 4) ? queueIndex + 1 : 4;
                        currentProcess->currentQueueLevel = nextLevel;
                        currentProcess->timeSpentInCurrentQueue = 0;
                        currentProcess->consecutiveSchedules = 0;
                        queues[nextLevel].push(currentProcess);

                        cout << "  " << currentProcess->name << " 时间片用完，降级到队列" << (nextLevel + 1) << endl;
                    }
                } else {
                    // 时间片未用完，重新加入当前队列
                    queues[queueIndex].push(currentProcess);
                    cout << "  " << currentProcess->name << " 时间片未用完，重新加入队列" << (queueIndex + 1) << endl;
                }

                displayQueueStatus();
            }
        }

        // 所有进程完成或达到最大时间
        cout << "\n" << string(100, '=') << endl;
        if (completedProcesses == totalProcesses) {
            cout << "所有进程已完成调度!" << endl;
        } else {
            cout << "达到最大时间限制 " << maxTime << "，调度提前结束!" << endl;
        }
        cout << string(100, '=') << endl;

        printFinalStatistics();
    }

    // 打印最终统计信息
    void printFinalStatistics() {
        cout << "\n最终统计信息:" << endl;
        cout << "名称\tPID\t到达\t服务\t开始\t完成\t周转\t等待" << endl;
        cout << "----\t---\t----\t----\t----\t----\t----\t----" << endl;

        double totalTurnaround = 0;
        double totalWaiting = 0;

        for (int i = 0; i < processes.size(); i++) {
            const Process& p = processes[i];

            int waiting = turnaround - p.burstTime;

                 << p.arriveTime << "\t" << p.burstTime << "\t"
                 << p.startTime << "\t" << p.finishTime << "\t"
                 << turnaround << "\t" << waiting << endl;
        }

        cout << "\n平均周转时间: " << fixed << setprecision(2) << (totalTurnaround / totalProcesses) << endl;
        cout << "平均等待时间: " << fixed << setprecision(2) << (totalWaiting / totalProcesses) << endl;
        cout << "总调度时间: " << currentTime << " 单位时间" << endl;

        // 显示队列配置
        cout << "\n队列配置总结:" << endl;
        for (int i = 0; i < 5; i++) {
            cout << "队列" << (i + 1) << ": 时间片大小 = " << timeSlices[i]

        }
    }
